cmake_minimum_required(VERSION 3.20)
project(fastsim_native_frontend VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The first native worker/frontend implementation currently targets Linux")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
option(FASTSIM_BUILD_GUI "Build Dear ImGui desktop frontend" ON)
add_library(fastsim_jobs native/frontend/native_job.cpp native/frontend/dps_histogram.cpp)
target_include_directories(fastsim_jobs PUBLIC native/frontend)
target_link_libraries(fastsim_jobs PUBLIC Threads::Threads nlohmann_json::nlohmann_json)
target_compile_options(fastsim_jobs PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(fastsim-job-test validation/native_job.cpp)
target_link_libraries(fastsim-job-test PRIVATE fastsim_jobs)
enable_testing()
find_program(FASTSIM_LUAJIT luajit)
if(FASTSIM_LUAJIT)
  add_test(NAME neuroevolution COMMAND "${FASTSIM_LUAJIT}" validation/neuroevolution.lua)
  set_tests_properties(neuroevolution PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME mcts_search COMMAND "${FASTSIM_LUAJIT}" validation/mcts.lua)
  set_tests_properties(mcts_search PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME regret_search COMMAND "${FASTSIM_LUAJIT}" validation/regret.lua)
  set_tests_properties(regret_search PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
add_test(NAME native_job COMMAND fastsim-job-test)
add_executable(fastsim-histogram-test validation/dps_histogram.cpp)
target_link_libraries(fastsim-histogram-test PRIVATE fastsim_jobs)
add_test(NAME dps_histogram COMMAND fastsim-histogram-test)
if(FASTSIM_BUILD_GUI)
  set(_frontend_sources tools/neuro_shader.lua tools/neuroevolution.lua tools/mcts.lua tools/policy_shader.lua tools/policy_controller.lua tools/run_mcts.lua tools/regret_transport.lua tools/regret.lua tools/run_regret.lua tools/detailed_trace.lua tools/detailed_analysis.lua native/ui/detailed_view.cpp native/ui/detailed_view.h native/frontend/dps_histogram.cpp native/frontend/dps_histogram.h native/ui/editor.cpp native/ui/editor.h tools/editor_config.mjs native/ui/main.cpp native/frontend/native_job.cpp native/frontend/native_job.h CMakeLists.txt third_party/imgui/SHA256SUMS third_party/implot/SHA256SUMS)
  file(GLOB_RECURSE _editor_inputs CONFIGURE_DEPENDS policies/warlock/source/* assets/icons/*)
  list(APPEND _frontend_sources ${_editor_inputs})
  set(_frontend_hashes "")
  foreach(_source IN LISTS _frontend_sources)
    if(IS_ABSOLUTE "${_source}")
      set(_source_path "${_source}")
    else()
      set(_source_path "${CMAKE_CURRENT_SOURCE_DIR}/${_source}")
    endif()
    file(SHA256 "${_source_path}" _source_hash)
    string(APPEND _frontend_hashes "${_source_hash}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_source_path}")
  endforeach()
  string(SHA256 _frontend_hash "${_frontend_hashes}")
  find_package(SDL3 3.2 REQUIRED CONFIG)
  find_package(OpenGL REQUIRED)
  find_package(PNG REQUIRED)
  find_program(FASTSIM_NODE node REQUIRED)
  add_test(NAME editor_config COMMAND "${FASTSIM_NODE}" validation/editor_config.mjs)
  set_tests_properties(editor_config PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_library(imgui STATIC
    third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
    third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
    third_party/imgui/backends/imgui_impl_sdl3.cpp third_party/imgui/backends/imgui_impl_opengl3.cpp
    third_party/implot/implot.cpp third_party/implot/implot_items.cpp)
  target_include_directories(imgui PUBLIC third_party/imgui third_party/imgui/backends third_party/implot)
  target_link_libraries(imgui PUBLIC SDL3::SDL3 OpenGL::GL ${CMAKE_DL_LIBS})
  add_executable(fastsim-imgui native/ui/main.cpp native/ui/editor.cpp native/ui/detailed_view.cpp)
  target_link_libraries(fastsim-imgui PRIVATE fastsim_jobs imgui PNG::PNG)
  target_compile_options(fastsim-imgui PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_definitions(fastsim-imgui PRIVATE FASTSIM_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" FASTSIM_IMGUI_REVISION="f5f6ca07be7ce0ea9eed6c04d55833bac3f6b50b" FASTSIM_FRONTEND_SOURCE_HASH="${_frontend_hash}" FASTSIM_FRONTEND_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
endif()
